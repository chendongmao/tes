```sql
-- subject     areas: water assets areas
-- function describe: count the number of water meters each year
-- create       date: 2025/11/19
-- modify date     modify by      modify content 
-- 
-- source table
-- coss_dwd.dwd_ass_user_meter_di                                     water meter information table         
-- target table
-- coss_dm.dm_ass_annual_pipe_asset_region_item_di               annual index information of pipeline network assets in the region
insert into coss_dm.dm_ass_annual_pipe_asset_region_item_di(
    statistical_year,          -- statistical year
    region_abbr,               -- regional abbreviation
    inter_item_code,           -- internal item code
    item_value                 -- item value
)
-- count the number of water meters in each operation area
select
    to_char(current_timestamp, 'YYYY') as statistical_year,
	region_abbr,
	'PN_AS_000001' as inter_item_code,
	count(1) as item_value
from
    coss_dwd.dwd_ass_user_meter_di
where meter_status in ('Active', 'Reconditioned', 'Meter is registering') and region_abbr is not null
group by
    statistical_year,
	region_abbr
union all
-- count the number of water meters in HKSAR
select
    to_char(current_timestamp, 'YYYY') as statistical_year,
	'HKSAR' as region_abbr,
	'PN_AS_000001' as inter_item_code,
	count(1) as item_value
from
    coss_dwd.dwd_ass_user_meter_di
where meter_status in ('Active', 'Reconditioned', 'Meter is registering')
group by
    statistical_year	
union all
-- count the total number of tables of master meter in each operation area
select
    to_char(current_timestamp, 'YYYY') as statistical_year,
	region_abbr,
	'PN_AS_000004' as inter_item_code,
	0 as item_value
from
    coss_dwd.dwd_ass_user_meter_di
where meter_status in ('Active', 'Reconditioned', 'Meter is registering') and region_abbr is not null
group by
    statistical_year,
	region_abbr
union all
-- count the total number of tables of master meter in HKSAR
select
    to_char(current_timestamp, 'YYYY') as statistical_year,
	'HKSAR' as region_abbr,
	'PN_AS_000004' as inter_item_code,
	0 as item_value
from
    coss_dwd.dwd_ass_user_meter_di
where meter_status in ('Active', 'Reconditioned', 'Meter is registering')
group by
    statistical_year
on duplicate key 
update
    item_value = values(item_value),
	dm_update_time = current_timestamp;
```

## 计算历史水表个数逻辑（年、运作区维度）

```sql
create table if not exists coss_dm.dm_ass_annual_pipe_asset_region_item_di_tmp (
	statistical_year varchar(4),
	region_abbr varchar(5),
	inter_item_code varchar(12),
	item_value float8,
    dm_load_time timestamp(6) default current_timestamp,
	dm_update_time timestamp(6) default current_timestamp,
    primary key (statistical_year,region_abbr,inter_item_code)
)
with (orientation = row, compression = no)
distribute by hash (statistical_year,region_abbr);
comment on table coss_dm.dm_ass_annual_pipe_asset_region_item_di_tmp is 'Annual index information of pipeline network assets in the region';
comment on column coss_dm.dm_ass_annual_pipe_asset_region_item_di_tmp.statistical_year is 'Statistical Year';
comment on column coss_dm.dm_ass_annual_pipe_asset_region_item_di_tmp.region_abbr is 'Regional Abbreviation';
comment on column coss_dm.dm_ass_annual_pipe_asset_region_item_di_tmp.inter_item_code is 'Inter Item Code';
comment on column coss_dm.dm_ass_annual_pipe_asset_region_item_di_tmp.item_value is 'Item Value';
comment on column coss_dm.dm_ass_annual_pipe_asset_region_item_di_tmp.dm_update_time is 'Data Update Time';
comment on column coss_dm.dm_ass_annual_pipe_asset_region_item_di_tmp.dm_load_time is 'Data Loading Time';

truncate table coss_dm.dm_ass_annual_pipe_asset_region_item_di_tmp;

-- 按运作区
with all_months as (
    select
	    'A' as auxiliary,
        generate_series as month_start
    from generate_series (
        '2000-01-01'::date,                      -- start month
        date_trunc('month', current_date)::date, -- end month
        interval '1 month'                       -- step length
    ) as generate_series
), all_dims as (
    select
        b.month_start,
	    a.region_abbr
    from (
	-- obtain region information
        select 
            'A' as auxiliary,
	        region_abbr
        from
            coss_dim.dim_region_info
        where region_abbr != 'HKSAR'
        union all
        select 'A' as auxiliary,'XXXX' as region_abbr
    ) a
	inner join all_months b
	on a.auxiliary = b.auxiliary
), all_meter_count as (
    select
	    date_trunc('month', receive_time)::date as month_start,
		region_abbr,
		count(*) as all_total
    from (
	    select
		    nvl(region_abbr, 'XXXX') as region_abbr,
			least(nvl(recond_time,'9999-01-01 00:00:00'), nvl(retire_time,'9999-01-01 00:00:00'), nvl(receive_time,'9999-01-01 00:00:00')) as receive_time
		from
	        coss_dwd.dwd_ass_user_meter_di
		where recond_time is not null or retire_time is not null or receive_time is not null
	) a
	where receive_time >= '2000-01-01 00:00:00'
	group by
		region_abbr,
		date_trunc('month', receive_time)
), all_meter_count_null as (
    select
	    nvl(region_abbr, 'XXXX') as region_abbr,
		count(*) as all_total_null
    from
	    coss_dwd.dwd_ass_user_meter_di
    where (recond_time is null and retire_time is null and receive_time is null) 
	    or (least(nvl(recond_time,'9999-01-01 00:00:00'), nvl(retire_time,'9999-01-01 00:00:00'), nvl(receive_time,'9999-01-01 00:00:00')) < '2000-01-01 00:00:00')
	group by nvl(region_abbr, 'XXXX')
), all_filled_counts as (
    select
        a.month_start,
		a.region_abbr,
        b.all_total
    from 
	    all_dims a
    left join
	    all_meter_count b
    on a.month_start = b.month_start and a.region_abbr = b.region_abbr
), all_meter as (
    select
        to_char(a.month_start, 'YYYYMM') as statistical_month,
		a.region_abbr,
        coalesce(
			sum(a.all_total) over (partition by a.region_abbr order by a.month_start) + coalesce(b.all_total_null, 0), 
			coalesce(b.all_total_null, 0)
		) as all_count
    from 
	    all_filled_counts a
    left join all_meter_count_null b
	on a.region_abbr = b.region_abbr
), retired_meter_counts as (                             
    select
	    date_trunc('month', retire_time)::date as month_start,
		region_abbr,
		count(*) as retired_total
    from (
	    select
		    nvl(region_abbr, 'XXXX') as region_abbr,
			greatest(nvl(retire_time,'1900-01-01 00:00:00'), nvl(receive_time,'1900-01-01 00:00:00')) as retire_time
		from
	        coss_dwd.dwd_ass_user_meter_di
        where meter_status = 'Retired' and (retire_time is not null or receive_time is not null)
		    and (greatest(nvl(retire_time,'1900-01-01 00:00:00'), nvl(receive_time,'1900-01-01 00:00:00')) >= '2000-01-01 00:00:00')
	) a
	group by
		region_abbr,
		date_trunc('month', retire_time)
), retired_meter_counts_null as (
    select
	    nvl(region_abbr, 'XXXX') as region_abbr,
		count(*) as retired_total_null
    from
	    coss_dwd.dwd_ass_user_meter_di
    where meter_status = 'Retired' and ((receive_time is null and retire_time is null)  or 
	    (greatest(nvl(retire_time,'1900-01-01 00:00:00'), nvl(receive_time,'1900-01-01 00:00:00')) < '2000-01-01 00:00:00'))
	group by nvl(region_abbr, 'XXXX')
), retired_filled_counts as (
    select
        a.month_start,
		a.region_abbr,
        b.retired_total
    from 
	    all_dims a
    left join
	    retired_meter_counts b
    on a.month_start = b.month_start and a.region_abbr = b.region_abbr
), retired_meter as (
    select
        to_char(a.month_start, 'YYYYMM') as statistical_month,
		a.region_abbr,
        coalesce(
			sum(a.retired_total) over (partition by a.region_abbr order by a.month_start) + coalesce(b.retired_total_null, 0), 
			coalesce(b.retired_total_null, 0)
		) as retired_count
    from 
	    retired_filled_counts a
    left join retired_meter_counts_null b
	on a.region_abbr = b.region_abbr
), recond_meter_counts as (
    select
	    date_trunc('month', recond_time)::date as month_start,
		region_abbr,
		count(*) as recond_total
    from (
	    select
		    nvl(region_abbr, 'XXXX') as region_abbr,
			greatest(nvl(recond_time,'1900-01-01 00:00:00'), nvl(retire_time,'1900-01-01 00:00:00'), nvl(receive_time,'1900-01-01 00:00:00')) as recond_time
		from
	        coss_dwd.dwd_ass_user_meter_di
		where meter_status = 'Reconditioned' and (recond_time is not null or retire_time is not null or receive_time is not null)
	) a
	where recond_time >= '2000-01-01 00:00:00'
	group by
		region_abbr,
		date_trunc('month', recond_time)
), recond_meter_counts_null as (
    select
	    nvl(region_abbr, 'XXXX') as region_abbr,
		count(*) as recond_total_null
    from
	    coss_dwd.dwd_ass_user_meter_di
    where meter_status = 'Reconditioned' and ((recond_time is null and retire_time is null and receive_time is null) 
	    or (greatest(nvl(recond_time,'1900-01-01 00:00:00'), nvl(retire_time,'1900-01-01 00:00:00'), nvl(receive_time,'1900-01-01 00:00:00')) < '2000-01-01 00:00:00'))
	group by nvl(region_abbr, 'XXXX')
), recond_filled_counts as (
    select
        a.month_start,
		a.region_abbr,
        b.recond_total
    from 
	    all_dims a
    left join
	    recond_meter_counts b
    on a.month_start = b.month_start and a.region_abbr = b.region_abbr
), recond_meter as (
    select
        to_char(a.month_start, 'YYYYMM') as statistical_month,
		a.region_abbr,
        coalesce(
			sum(a.recond_total) over (partition by a.region_abbr order by a.month_start) + coalesce(b.recond_total_null, 0), 
			coalesce(b.recond_total_null, 0)
		) as recond_count
    from 
	    recond_filled_counts a
    left join recond_meter_counts_null b
	on a.region_abbr = b.region_abbr
), sample_meter_counts as (
    select
	    date_trunc('month', receive_time)::date as month_start,
		nvl(region_abbr, 'XXXX') as region_abbr,
		count(*) as sample_total
    from
	    coss_dwd.dwd_ass_user_meter_di
	where meter_status in ('Meter is not registering', 'Pending for Sample Inspection', 'Sample Inspection Rejected') and receive_time is not null 
	    and receive_time >= '2000-01-01 00:00:00'
	group by
		nvl(region_abbr, 'XXXX'),
		date_trunc('month', receive_time)
), sample_meter_counts_null as (
    select
	    nvl(region_abbr, 'XXXX') as region_abbr,
		count(*) as sample_total_null
    from
	    coss_dwd.dwd_ass_user_meter_di
    where meter_status in ('Meter is not registering', 'Pending for Sample Inspection', 'Sample Inspection Rejected') 
	    and (receive_time is null or receive_time < '2000-01-01 00:00:00')
	group by nvl(region_abbr, 'XXXX')
), sample_filled_counts as (
    select
        a.month_start,
		a.region_abbr,
        b.sample_total
    from 
	    all_dims a
    left join
	    sample_meter_counts b
    on a.month_start = b.month_start and a.region_abbr = b.region_abbr
), sample_meter as (
    select
        to_char(a.month_start, 'YYYYMM') as statistical_month,
		a.region_abbr,
        coalesce(
			sum(a.sample_total) over (partition by a.region_abbr order by a.month_start) + coalesce(b.sample_total_null, 0), 
			coalesce(b.sample_total_null, 0)
		) as sample_count
    from 
	    sample_filled_counts a
    left join sample_meter_counts_null b
	on a.region_abbr = b.region_abbr
), month_meter as (
    select
	    a.statistical_month,
		a.region_abbr,
	    (a.all_count - b.retired_count + c.recond_count - d.sample_count) as month_meter
	from
	    all_meter a
	inner join 
	    retired_meter b
	on a.statistical_month = b.statistical_month and a.region_abbr = b.region_abbr
	inner join recond_meter c
	on a.statistical_month = c.statistical_month and a.region_abbr = c.region_abbr
    inner join sample_meter d
	on a.statistical_month = d.statistical_month and a.region_abbr = d.region_abbr
)

insert into coss_dm.dm_ass_annual_pipe_asset_region_item_di_tmp(
    statistical_year,          -- statistical year
    region_abbr,               -- regional abbreviation
    inter_item_code,           -- internal item code
    item_value                 -- item value
)
select 
    substring(statistical_month,1,4) as statistical_year,
	region_abbr,
	'PN_AS_000001' as inter_item_code,
	month_meter as item_value
from
    month_meter
where substring(statistical_month,5,2) = 12 or statistical_month = to_char(current_timestamp, 'YYYYMM');

insert into coss_dm.dm_ass_annual_pipe_asset_region_item_di(
    statistical_year,          -- statistical year
    region_abbr,               -- regional abbreviation
    inter_item_code,           -- internal item code
    item_value                 -- item value
)
select 
    statistical_year,
	region_abbr,
	inter_item_code,
	item_value
from
    coss_dm.dm_ass_annual_pipe_asset_region_item_di_tmp
union all 
select 
    statistical_year,
	'HKSAR' as region_abbr,
	inter_item_code,
	sum(item_value) as item_value
from
    coss_dm.dm_ass_annual_pipe_asset_region_item_di_tmp
group by statistical_year,inter_item_code
on duplicate key 
update
    item_value = values(item_value),
	dm_update_time = current_timestamp;
```

```sql
-- subject     areas: water assets areas
-- function describe: calculate historical stock of water meters by year / region / meter type
-- create       date: 2026/08/14
-- modify date     modify by      modify content 
-- 
-- source table
-- coss_dwd.dwd_ass_user_meter_di                                     water meter information table         
-- target table
-- coss_dm.dm_tmu_annual_user_customer_item_stg_di                   Annual index of user meter by region & meter type

-- coss_dm.dm_tmu_annual_user_customer_item_stg_di_tmp              Temporary staging table for historical meter stock calculation

-- Step1 创建临时中转表
create table if not exists coss_dm.dm_tmu_annual_user_customer_item_stg_di_tmp (
	statistical_year varchar(4),
	region_abbr varchar(5),
	inter_item_code varchar(12),
	type_code varchar(12),
	item_value float8,
    dm_load_time timestamp(6) default current_timestamp,
	dm_update_time timestamp(6) default current_timestamp,
    primary key (statistical_year,region_abbr,inter_item_code,type_code)
)
with (orientation = row, compression = no)
distribute by hash (statistical_year,region_abbr,type_code);

-- 临时表字段注释
comment on table coss_dm.dm_tmu_annual_user_customer_item_stg_di_tmp is 'Temporary table for historical user meter index by region & meter type';
comment on column coss_dm.dm_tmu_annual_user_customer_item_stg_di_tmp.statistical_year is 'Statistical Year';
comment on column coss_dm.dm_tmu_annual_user_customer_item_stg_di_tmp.region_abbr is 'Regional Abbreviation';
comment on column coss_dm.dm_tmu_annual_user_customer_item_stg_di_tmp.inter_item_code is 'Internal Item Code';
comment on column coss_dm.dm_tmu_annual_user_customer_item_stg_di_tmp.type_code is 'Meter Type Code';
comment on column coss_dm.dm_tmu_annual_user_customer_item_stg_di_tmp.item_value is 'Index Value';
comment on column coss_dm.dm_tmu_annual_user_customer_item_stg_di_tmp.dm_load_time is 'Data Loading Time';
comment on column coss_dm.dm_tmu_annual_user_customer_item_stg_di_tmp.dm_update_time is 'Data Update Time';

-- 清空临时表，全量重跑
truncate table coss_dm.dm_tmu_annual_user_customer_item_stg_di_tmp;

-- Step2 CTE时序分层计算（新增type_code水表类型拆分，口径与原历史存量完全对齐）
with all_months as (
    select
	    'A' as auxiliary,
        generate_series as month_start
    from generate_series (
        '2000-01-01'::date,                      -- start month
        date_trunc('month', current_date)::date, -- end month
        interval '1 month'                       -- step length
    ) as generate_series
), all_region_dim as (
	-- 获取标准区域，排除HKSAR，新增未知区域XXXX
    select 
        'A' as auxiliary,
	    region_abbr
    from coss_dim.dim_region_info
    where region_abbr != 'HKSAR'
    union all
    select 'A' as auxiliary,'XXXX' as region_abbr
), all_type_dim as (
	-- 水表类型维度：普通表、远传表，用于补齐维度笛卡尔积
	select 'ME_TY_000001' as type_code union select 'ME_TY_000002' as type_code
), all_dims as (
    select
        b.month_start,
	    a.region_abbr,
		c.type_code
    from all_region_dim a
	inner join all_months b
	on a.auxiliary = b.auxiliary
	cross join all_type_dim c
), all_meter_source as (
	-- 基础水表数据源，拆分区域、水表类型、业务时间字段
	select
		nvl(region_abbr, 'XXXX') as region_abbr,
		case when amr = 'Y' then 'ME_TY_000002' else 'ME_TY_000001' end as type_code,
		least(nvl(recond_time,'9999-01-01 00:00:00'), nvl(retire_time,'9999-01-01 00:00:00'), nvl(receive_time,'9999-01-01 00:00:00')) as receive_time,
		greatest(nvl(retire_time,'1900-01-01 00:00:00'), nvl(receive_time,'1900-01-01 00:00:00')) as retire_time,
		greatest(nvl(recond_time,'1900-01-01 00:00:00'), nvl(retire_time,'1900-01-01 00:00:00'), nvl(receive_time,'1900-01-01 00:00:00')) as recond_time,
		meter_status
	from coss_dwd.dwd_ass_user_meter_di
), all_meter_count as (
    select
	    date_trunc('month', receive_time)::date as month_start,
		region_abbr,
		type_code,
		count(distinct meter_no) as all_total
    from all_meter_source
	where recond_time is not null or retire_time is not null or receive_time is not null
	  and receive_time >= '2000-01-01 00:00:00'
	group by region_abbr, type_code, date_trunc('month', receive_time)
), all_meter_count_null as (
    select
	    region_abbr,
		type_code,
		count(distinct meter_no) as all_total_null
    from all_meter_source
    where (recond_time is null and retire_time is null and receive_time is null) 
	    or receive_time < '2000-01-01 00:00:00'
	group by region_abbr, type_code
), all_filled_counts as (
    select
        a.month_start,
		a.region_abbr,
		a.type_code,
        b.all_total
    from all_dims a
    left join all_meter_count b
    on a.month_start = b.month_start and a.region_abbr = b.region_abbr and a.type_code = b.type_code
), all_meter as (
    select
        to_char(a.month_start, 'YYYYMM') as statistical_month,
		a.region_abbr,
		a.type_code,
        coalesce(
			sum(a.all_total) over (partition by a.region_abbr,a.type_code order by a.month_start) + coalesce(b.all_total_null, 0), 
			coalesce(b.all_total_null, 0)
		) as all_count
    from all_filled_counts a
    left join all_meter_count_null b
	on a.region_abbr = b.region_abbr and a.type_code = b.type_code
), retired_meter_counts as (                             
    select
	    date_trunc('month', retire_time)::date as month_start,
		region_abbr,
		type_code,
		count(distinct meter_no) as retired_total
    from all_meter_source
    where meter_status = 'Retired' and (retire_time is not null or receive_time is not null)
	  and retire_time >= '2000-01-01 00:00:00'
	group by region_abbr, type_code, date_trunc('month', retire_time)
), retired_meter_counts_null as (
    select
	    region_abbr,
		type_code,
		count(distinct meter_no) as retired_total_null
    from all_meter_source
    where meter_status = 'Retired' 
	  and ((receive_time is null and retire_time is null) or retire_time < '2000-01-01 00:00:00')
	group by region_abbr, type_code
), retired_filled_counts as (
    select
        a.month_start,
		a.region_abbr,
		a.type_code,
        b.retired_total
    from all_dims a
    left join retired_meter_counts b
    on a.month_start = b.month_start and a.region_abbr = b.region_abbr and a.type_code = b.type_code
), retired_meter as (
    select
        to_char(a.month_start, 'YYYYMM') as statistical_month,
		a.region_abbr,
		a.type_code,
        coalesce(
			sum(a.retired_total) over (partition by a.region_abbr,a.type_code order by a.month_start) + coalesce(b.retired_total_null, 0), 
			coalesce(b.retired_total_null, 0)
		) as retired_count
    from retired_filled_counts a
    left join retired_meter_counts_null b
	on a.region_abbr = b.region_abbr and a.type_code = b.type_code
), recond_meter_counts as (
    select
	    date_trunc('month', recond_time)::date as month_start,
		region_abbr,
		type_code,
		count(distinct meter_no) as recond_total
    from all_meter_source
	where meter_status = 'Reconditioned' and (recond_time is not null or retire_time is not null or receive_time is not null)
	  and recond_time >= '2000-01-01 00:00:00'
	group by region_abbr, type_code, date_trunc('month', recond_time)
), recond_meter_counts_null as (
    select
	    region_abbr,
		type_code,
		count(distinct meter_no) as recond_total_null
    from all_meter_source
    where meter_status = 'Reconditioned' 
	  and ((recond_time is null and retire_time is null and receive_time is null) or recond_time < '2000-01-01 00:00:00')
	group by region_abbr, type_code
), recond_filled_counts as (
    select
        a.month_start,
		a.region_abbr,
		a.type_code,
        b.recond_total
    from all_dims a
    left join recond_meter_counts b
    on a.month_start = b.month_start and a.region_abbr = b.region_abbr and a.type_code = b.type_code
), recond_meter as (
    select
        to_char(a.month_start, 'YYYYMM') as statistical_month,
		a.region_abbr,
		a.type_code,
        coalesce(
			sum(a.recond_total) over (partition by a.region_abbr,a.type_code order by a.month_start) + coalesce(b.recond_total_null, 0), 
			coalesce(b.recond_total_null, 0)
		) as recond_count
    from recond_filled_counts a
    left join recond_meter_counts_null b
	on a.region_abbr = b.region_abbr and a.type_code = b.type_code
), sample_meter_counts as (
    select
	    date_trunc('month', receive_time)::date as month_start,
		region_abbr,
		type_code,
		count(distinct meter_no) as sample_total
    from all_meter_source
	where meter_status in ('Meter is not registering', 'Pending for Sample Inspection', 'Sample Inspection Rejected') 
	  and receive_time is not null and receive_time >= '2000-01-01 00:00:00'
	group by region_abbr, type_code, date_trunc('month', receive_time)
), sample_meter_counts_null as (
    select
	    region_abbr,
		type_code,
		count(distinct meter_no) as sample_total_null
    from all_meter_source
    where meter_status in ('Meter is not registering', 'Pending for Sample Inspection', 'Sample Inspection Rejected') 
	  and (receive_time is null or receive_time < '2000-01-01 00:00:00')
	group by region_abbr, type_code
), sample_filled_counts as (
    select
        a.month_start,
		a.region_abbr,
		a.type_code,
        b.sample_total
    from all_dims a
    left join sample_meter_counts b
    on a.month_start = b.month_start and a.region_abbr = b.region_abbr and a.type_code = b.type_code
), sample_meter as (
    select
        to_char(a.month_start, 'YYYYMM') as statistical_month,
		a.region_abbr,
		a.type_code,
        coalesce(
			sum(a.sample_total) over (partition by a.region_abbr,a.type_code order by a.month_start) + coalesce(b.sample_total_null, 0), 
			coalesce(b.sample_total_null, 0)
		) as sample_count
    from sample_filled_counts a
    left join sample_meter_counts_null b
	on a.region_abbr = b.region_abbr and a.type_code = b.type_code
), month_meter as (
    select
	    a.statistical_month,
		a.region_abbr,
		a.type_code,
	    (a.all_count - b.retired_count + c.recond_count - d.sample_count) as month_meter
	from all_meter a
	inner join retired_meter b
	on a.statistical_month = b.statistical_month and a.region_abbr = b.region_abbr and a.type_code = b.type_code
	inner join recond_meter c
	on a.statistical_month = c.statistical_month and a.region_abbr = c.region_abbr and a.type_code = c.type_code
    inner join sample_meter d
	on a.statistical_month = d.statistical_month and a.region_abbr = d.region_abbr and a.type_code = d.type_code
)

-- Step3 将年度存量写入临时表：取每年12月 or 当前年月作为年度存量值
insert into coss_dm.dm_tmu_annual_user_customer_item_stg_di_tmp(
    statistical_year,
    region_abbr,
    inter_item_code,
	type_code,
    item_value
)
select 
    substring(statistical_month,1,4) as statistical_year,
	region_abbr,
	'US_CM_000001' as inter_item_code,
	type_code,
	month_meter as item_value
from month_meter
where substring(statistical_month,5,2) = '12' or statistical_month = to_char(current_timestamp, 'YYYYMM');

-- Step4 临时表数据写入正式表，生成各区域明细 + HKSAR全港汇总行
insert into coss_dm.dm_tmu_annual_user_customer_item_stg_di(
    statistical_year,
    region_abbr,
    inter_item_code,
	type_code,
    item_value,
	dm_load_time,
	dm_update_time
)
-- 各运作区分类型明细
select 
    statistical_year,
	region_abbr,
	inter_item_code,
	type_code,
	item_value,
	current_timestamp as dm_load_time,
	current_timestamp as dm_update_time
from coss_dm.dm_tmu_annual_user_customer_item_stg_di_tmp
union all 
-- HKSAR全港汇总：同一年份+同一指标+同一水表类型求和
select 
    statistical_year,
	'HKSAR' as region_abbr,
	inter_item_code,
	type_code,
	sum(item_value) as item_value,
	current_timestamp as dm_load_time,
	current_timestamp as dm_update_time
from coss_dm.dm_tmu_annual_user_customer_item_stg_di_tmp
group by statistical_year, inter_item_code, type_code
on duplicate key 
update
    item_value = values(item_value),
	dm_update_time = current_timestamp;

-- Step5 删除未知区域XXXX，仅保留标准区域+全港汇总HKSAR
delete from coss_dm.dm_tmu_annual_user_customer_item_stg_di where region_abbr = 'XXXX';
```
